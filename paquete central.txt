  PROCEDURE P_VERIFLLAMADA_DBLINK(pDBLink     IN VARCHAR2,
                                  pSitio      IN NUMBER,
                                  pNumero     IN NUMBER,
                                  pQuery      OUT varchar2,
                                  pCantidad   OUT NUMBER,
                                  pcodsalida  OUT VARCHAR2,
                                  pTEXTsalida OUT VARCHAR2) is
    TYPE regcons IS RECORD(
      cantRegistros NUMBER(12));
    TYPE pCursorConsec IS REF CURSOR;
    CantRegistros pCursorConsec;
    vreg          regcons;
    vCant         NUMBER := 0;
  
    vSql  VARCHAR2(32767) := 'select nvl(count(*),0)';
    vSql2 VARCHAR2(32767) := ' from usr_cad.cadpedidos@' || pDBLink || ' a';
    vCSql VARCHAR2(32767) := ' where a.sitio_graba = ' || pSitio ||
                             ' and a.nume_llamada = ' || pNumero;
  
  begin
    --
    -- Calcular con el usuario el dato del sitio de grabacion
    BEGIN
      -- calcular el dato del consecutivo remoto
      PCODSALIDA  := '003'; --Generando query para conectarse a la base de datos remota
      vSql        := vSql || vSql2 || vCSql;
      pTEXTsalida := vSql;
      PCODSALIDA  := '004'; --Extractando datos de la cantidad de registros existentes en la base de datos remota
      OPEN CantRegistros FOR vSql;
      LOOP
        FETCH CantRegistros
          INTO vreg;
        EXIT WHEN CantRegistros%NOTFOUND;
        vCant := vreg.cantRegistros;
      END LOOP;
      CLOSE CantRegistros;
      PCODSALIDA := '005'; --Retornando los datos de la cantidad de registros existentes en la base de datos remota
      pQuery     := vSql;
      pCantidad  := vCant;
      if pCantidad = 0 then
        pTEXTsalida := 'No se inserto el registro. ';
      else
        pTEXTsalida := 'Registro Ok. ';
      end if;
    EXCEPTION
      WHEN OTHERS THEN
        pQuery      := vSql || '. ' || SQLERRM;
        pCantidad   := 0;
        pcodsalida  := sqlcode;
        pTEXTsalida := 'No se inserto el registro. ' || SQLERRM;
    END;
  end P_VERIFLLAMADA_DBLINK;
